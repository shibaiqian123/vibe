cmake_minimum_required(VERSION 3.18)
project(vibe_trt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_path(TRT_INCLUDE NvInfer.h HINTS "$ENV{TensorRT_ROOT}/include" /usr/include/x86_64-linux-gnu)
find_library(TRT_LIBRARY nvinfer HINTS "$ENV{TensorRT_ROOT}/lib")
find_library(TRT_PARSER nvonnxparser HINTS "$ENV{TensorRT_ROOT}/lib")
if(NOT TRT_INCLUDE OR NOT TRT_LIBRARY OR NOT TRT_PARSER)
  message(FATAL_ERROR "TensorRT headers/libraries missing; existing deployment installation required")
endif()
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/sonic/gear_sonic_deploy/src/TRTInference")
# Keep vendored upstream byte-identical; document these two build-time changes.
file(READ "${UPSTREAM}/InferenceEngine.cpp" ENGINE_SOURCE)
string(REPLACE "config->addOptimizationProfile( profile );" "config->addOptimizationProfile( profile );\n    config->clearFlag(nvinfer1::BuilderFlag::kTF32);" ENGINE_SOURCE "${ENGINE_SOURCE}")
string(REPLACE "// name of the tensorrt" "trtHash = picosha2::hash256_hex_string(trtHash + std::to_string(NV_TENSORRT_VERSION) + std::string(\"vibe-fp32-no-tf32-v1\"));\n    // name of the tensorrt" ENGINE_SOURCE "${ENGINE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/InferenceEngine.cpp" "${ENGINE_SOURCE}")
add_library(vibe_trt SHARED trt_bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/InferenceEngine.cpp")
target_include_directories(vibe_trt PRIVATE "${UPSTREAM}" "${TRT_INCLUDE}")
target_link_libraries(vibe_trt PRIVATE CUDA::cudart "${TRT_LIBRARY}" "${TRT_PARSER}")
# Deliberately no -ffast-math; numerical gate stays at 1e-5.
